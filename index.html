            // Returns minutes from now until the latest start time of a task.

            function minutesUntilLatestStart(task) {
                const now = new Date();
                const day = now.getDay(); // 0=Sun, 1=Mon, ..., 5=Fri, 6=Sat
                const h = now.getHours();
                const m = now.getMinutes();
                const nowMin = h * 60 + m;

                const latestStartOffset = task.deadline - task.duration;

                let currentOffset = null;

                if (day === 5 && h >= 19) {
                    // Friday after 19:00
                    currentOffset = nowMin - 19 * 60;
                } else if (day === 6) {
                    // Saturday — 5 hours from Fri 19:00 to Sat 00:00 = 300 min
                    currentOffset = 5 * 60 + nowMin;
                } else if (day === 0 && h < 16) {
                    // Sunday before 16:00 — 29 hours from Fri 19:00 to Sun 00:00 = 1740 min
                    currentOffset = 29 * 60 + nowMin;
                }

                if (currentOffset !== null) {
                    return latestStartOffset - currentOffset;
                }

                // Outside the window — compute distance to next Friday 19:00
                let daysToFri = (5 - day + 7) % 7;
                if (daysToFri === 0 && h >= 19) daysToFri = 7;
                const minutesToFri19 = daysToFri * 24 * 60 + (19 * 60 - nowMin);
                return minutesToFri19 + latestStartOffset;
            }

            function parseTimeToHours(str) {
                str = str.trim().toLowerCase();
                if (str.endsWith('h')) {
                    const val = parseFloat(str.replace('h', '').trim());
                    return isNaN(val) ? 0 : val;
                }
                if (str.endsWith('min')) {
                    const val = parseFloat(str.replace('min', '').trim());
                    return isNaN(val) ? 0 : val / 60;
                }
                const num = parseFloat(str);
                return isNaN(num) ? 0 : num;
            }

                if (m === 0) return h + 'h';
                return h + 'h ' + m + 'min';

            function compressImage(file, maxDim, quality) {
                return new Promise((resolve) => {
                    const reader = new FileReader();
                    reader.onload = (e) => {
                        const img = new Image();
                        img.onload = () => {
                            let { width, height } = img;
                            if (width > height && width > maxDim) {
                                height = height * (maxDim / width);
                                width = maxDim;
                            } else if (height > maxDim) {
                                width = width * (maxDim / height);
                                height = maxDim;
                            }
                            const canvas = document.createElement('canvas');
                            canvas.width = width;
                            canvas.height = height;
                            const ctx = canvas.getContext('2d');
                            ctx.drawImage(img, 0, 0, width, height);
                            resolve(canvas.toDataURL('image/jpeg', quality));
                        };
                        img.src = e.target.result;
                    };
                    reader.readAsDataURL(file);
                });
            }

            function bindWeekendActions() {
                document.querySelectorAll('[data-waction="toggle"]').forEach(btn => {
                    btn.addEventListener('click', function() {
                        const i = parseInt(this.dataset.wi);
                        if (i < 0 || i >= weekendTasks.length) return;
                        weekendTasks[i].completed = !weekendTasks[i].completed;
                        saveData();
                        renderWeekend();
                    });
                });
                document.querySelectorAll('[data-waction="delete"]').forEach(btn => {
                    btn.addEventListener('click', function() {
                        const i = parseInt(this.dataset.wi);
                        if (i < 0 || i >= weekendTasks.length) return;
                        if (confirm(`Delete "${weekendTasks[i].subject}"?`)) {
                            weekendTasks.splice(i, 1);
                            saveData();
                            renderWeekend();
                        }
                    });
                });
                document.querySelectorAll('[data-waction="upload"]').forEach(btn => {
                    btn.addEventListener('click', function() {
                        const i = parseInt(this.dataset.wi);
                        const input = document.querySelector(`.weekend-photo-input[data-wi="${i}"]`);
                        if (input) input.click();
                    });
                });
                document.querySelectorAll('.weekend-photo-input').forEach(input => {
                    input.addEventListener('change', async function(e) {
                        const i = parseInt(this.dataset.wi);
                        const file = e.target.files[0];
                        if (!file) return;
                        try {
                            const base64 = await compressImage(file, 800, 0.7);
                            if (!weekendTasks[i].photos) weekendTasks[i].photos = [];
                            weekendTasks[i].photos.push(base64);
                            saveData();
                            renderWeekend();
                        } catch (err) {
                            console.error(err);
                            alert('Failed to process image.');
                        }
                    });
                });
                document.querySelectorAll('.photo-thumb').forEach(img => {
                    img.addEventListener('click', function() {
                        openPhotoViewer(this.src);
                    });
                });
            }

            function addWeekendTask() {
                const subject = document.getElementById('weekendSubject').value.trim();
                const duration = parseInt(document.getElementById('weekendDuration').value) || 0;
                const slider = document.getElementById('deadlineSlider');
                const deadline = parseInt(slider.value) * 15;

                if (!subject) { alert('Please enter a subject'); return; }
                if (duration <= 0) { alert('Please enter duration in minutes (e.g. 60)'); return; }
                if (duration > WINDOW_TOTAL_MIN) { alert('Duration cannot exceed the whole weekend window'); return; }

                const latestStart = deadline - duration;
                if (latestStart < 0) {
                    if (!confirm('This task cannot be completed in time if it is not started immediately. Add anyway?')) return;
                }

                weekendTasks.push({
                    id: Date.now().toString(),
                    subject: subject,
                    duration: duration,
                    deadline: deadline,
                    completed: false,
                    photos: [],
                    createdAt: Date.now()
                });
                saveData();
                renderWeekend();
            }

            function setPage(index, animate = true) {
                if (index === currentPage && animate) return;
                currentPage = index;
                if (!animate) {
                    track.style.transition = 'none';
                } else {
                    track.style.transition = 'transform 0.4s cubic-bezier(0.22, 0.68, 0.35, 1)';
                }
                track.style.transform = `translateX(-${index * 100}%)`;
                dots.forEach((dot, i) => dot.classList.toggle('active', i === index));

                if (!animate) {
                    void track.offsetHeight;
                    track.style.transition = 'transform 0.4s cubic-bezier(0.22, 0.68, 0.35, 1)';
                }
            }

            const carousel = document.getElementById('carousel');
            let dragStartX = 0, dragStartY = 0;
            let isDragging = false, isHorizontal = false;
            let dragStartOffset = 0;

            function isInteractiveElement(el) {
                if (!el) return false;
                const tag = el.tagName.toLowerCase();
                if (tag === 'input' || tag === 'button' || tag === 'select' || tag === 'textarea') return true;
                if (el.isContentEditable) return true;
                if (el.closest && el.closest('input, button, select, textarea, .nt-slider, .photo-upload-btn, .supervisor-toggle, .log-delete-toggle')) return true;
                return false;
            }

            carousel.addEventListener('touchstart', function(e) {
                const touch = e.touches[0];
                const target = document.elementFromPoint(touch.clientX, touch.clientY);
                if (isInteractiveElement(target)) { isDragging = false; return; }
                dragStartX = touch.clientX;
                dragStartY = touch.clientY;
                isDragging = true;
                isHorizontal = false;
                dragStartOffset = -currentPage * 100;
                track.style.transition = 'none';
            }, { passive: true });

            carousel.addEventListener('touchmove', function(e) {
                if (!isDragging) return;
                const touch = e.touches[0];
                const dx = touch.clientX - dragStartX;
                const dy = touch.clientY - dragStartY;
                if (Math.abs(dx) > Math.abs(dy) && Math.abs(dx) > 10) {
                    isHorizontal = true;
                    e.preventDefault();
                    const dampened = dx * 0.6;
                    const offsetPercent = (dampened / carousel.offsetWidth) * 100;
                    let newOffset = dragStartOffset + offsetPercent;
                    const maxIndex = dots.length - 1;
                    if (newOffset > 0) newOffset = 0;
                    if (newOffset < -maxIndex * 100) newOffset = -maxIndex * 100;
                    track.style.transform = `translateX(${newOffset}%)`;
                }
            }, { passive: false });

            carousel.addEventListener('touchend', function(e) {
                if (!isDragging) return;
                isDragging = false;
                if (!isHorizontal) return;
                const dx = e.changedTouches[0].clientX - dragStartX;
                const threshold = 40;
                if (dx < -threshold && currentPage < dots.length - 1) setPage(currentPage + 1);
                else if (dx > threshold && currentPage > 0) setPage(currentPage - 1);
                else setPage(currentPage);
            }, { passive: true });

            let mouseDown = false, mouseStartX = 0, mouseStartOffset = 0;
            carousel.addEventListener('mousedown', function(e) {
                if (isInteractiveElement(e.target)) { mouseDown = false; return; }
                mouseDown = true;
                mouseStartX = e.clientX;
                mouseStartOffset = -currentPage * 100;
                track.style.transition = 'none';
                e.preventDefault();
            });
            document.addEventListener('mousemove', function(e) {
                if (!mouseDown) return;
                const dx = e.clientX - mouseStartX;
                const dampened = dx * 0.6;
                const offsetPercent = (dampened / carousel.offsetWidth) * 100;
                let newOffset = mouseStartOffset + offsetPercent;
                const maxIndex = dots.length - 1;
                if (newOffset > 0) newOffset = 0;
                if (newOffset < -maxIndex * 100) newOffset = -maxIndex * 100;
                track.style.transform = `translateX(${newOffset}%)`;
            });
            document.addEventListener('mouseup', function(e) {
                if (!mouseDown) return;
                mouseDown = false;
                const dx = e.clientX - mouseStartX;
                const threshold = 40;
                if (dx < -threshold && currentPage < dots.length - 1) setPage(currentPage + 1);
                else if (dx > threshold && currentPage > 0) setPage(currentPage - 1);
                else setPage(currentPage);
            });

            dots.forEach((dot, i) => {
                dot.addEventListener('click', function() { setPage(i); });
            });
